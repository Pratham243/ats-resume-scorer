# ATS Resume Scorer — combined backend + frontend dependencies.
#
# Install with:
#     pip install -r requirements.txt
#
# spaCy NLP models are installed below as direct wheel URLs so no
# `python -m spacy download` step is required at deploy time
# (Streamlit Cloud does not support post-install shell hooks).
#
# WeasyPrint (used for PDF export) also needs system libraries on Linux:
#     sudo dnf install -y cairo pango gdk-pixbuf2 libffi    # Fedora
#     sudo apt install -y libcairo2 libpango-1.0-0 libpangoft2-1.0-0 libffi-dev  # Debian/Ubuntu
# These are listed in packages.txt for Streamlit Cloud.

# ── Backend: web framework ──
fastapi>=0.110
uvicorn[standard]>=0.27
pydantic>=2.6
python-multipart>=0.0.9      # required by FastAPI for File/Form uploads
python-dotenv>=1.0

# ── Backend: NLP / ML ──
# numpy pinned <2.0: spaCy 3.7.x bundles thinc 8.2 which is NOT numpy-2
# compatible — importing spacy crashes at startup with "binary incompatibility"
# or "numpy.dtype size changed" errors. Pinning here forces a 1.x install
# everywhere so the subprocess (and tests) agree.
numpy>=1.26,<2.0
spacy>=3.7,<3.8              # 3.7.x is the latest line that's stable with thinc 8.2
sentence-transformers>=2.5,<3.0
rapidfuzz>=3.6

# ── spaCy models (installed as pip wheels, cached as normal Python packages) ──
https://github.com/explosion/spacy-models/releases/download/en_core_web_sm-3.7.1/en_core_web_sm-3.7.1-py3-none-any.whl
https://github.com/explosion/spacy-models/releases/download/en_core_web_md-3.7.1/en_core_web_md-3.7.1-py3-none-any.whl

# ── Backend: file parsing ──
pdfplumber>=0.11
PyPDF2>=3.0
python-docx>=1.1
python-magic>=0.4.27; sys_platform != 'win32'
python-magic-bin>=0.4.14; sys_platform == 'win32'

# ── Backend: PDF report generation ──
jinja2>=3.1
weasyprint>=60.0

# ── Backend: external services ──
httpx>=0.27                  # async Supabase REST client
groq>=0.13                   # Groq LLM client (0.13+ is required for httpx>=0.28 compatibility)
PyJWT[crypto]>=2.8           # verifies Supabase access tokens (HS256)

# ── Frontend ──
streamlit>=1.32
requests>=2.31
supabase>=2.4                # frontend auth client (email/password + OAuth)
